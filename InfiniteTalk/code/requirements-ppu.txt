# InfiniteTalk 在 PPU-ZW810E 上的补充依赖
#
# 用法（必须先装好厂商 PPU 版 torch / torchvision / torchaudio / xformers）：
#   uv pip install --python .venv-ppu -r requirements.txt -r requirements-ppu.txt
#
# 本文件只补 requirements.txt 里缺失的、或需要覆盖版本的包。
# 下面的注释说明了每个包「为什么需要」，便于排查。

# ---- 源码 import 了但 requirements.txt 未列出 ----

misaki[en]          # kokoro TTS 需要（from misaki import en, espeak）
                    # 必须带 [en] extra：否则缺 spacy / num2words，报 ModuleNotFoundError
espeakng-loader     # misaki 的 espeak 后端需要
phonemizer          # misaki 需要
protobuf            # transformers 转换 umt5 tokenizer 需要
                    # 缺了会报：Cannot parse precompiled_charsmap
tiktoken            # protobuf 缺失时的兜底路径（上一条的备选）
soundfile           # 音频读写（源码 import soundfile）
einops              # 张量重排（from einops import rearrange）
ftfy                # 文本清洗
triton==3.6.0       # xfuser 导入链需要（sparge_attention/block_mask.py import triton）
                    # 必须与 torch 2.10 配套，用 3.6.0

# ---- 版本约束（关键，不加会装错）----

# librosa 1.0+ 强制 numpy>=2，与 requirements.txt 的 numpy<2 冲突，
# 且会反向升级 numpy 破坏其它包。锁到 0.11.x。
librosa<1.0

# diffusers 会被解析到 0.40+，它要求 peft>=0.17；
# 不显式指定可能装到旧版 peft，导致：
#   ImportError: cannot import name 'disable_input_dtype_casting' from 'peft.helpers'
peft>=0.17
